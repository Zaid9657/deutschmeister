# AI and search crawlers are all welcome — Googlebot, Bingbot, GPTBot,
# ChatGPT-User, Claude-Web, anthropic-ai, PerplexityBot, Google-Extended,
# OAI-SearchBot, Applebot-Extended, CCBot, meta-externalagent, Bytespider and
# cohere-ai included. They used to be listed as 14 separate groups, each with a
# bare "Allow: /". Per RFC 9309 a crawler obeys ONLY the most specific group that
# matches it and ignores all others — so every one of those agents matched its own
# group and never saw the Disallow lines below. The single group grants exactly
# the same access and makes the exclusions actually bind.

User-agent: *
Allow: /
Disallow: /auth/
Disallow: /admin/
Disallow: /api/
# SPA shell — duplicate of the prerendered routes, never index directly
Disallow: /app.html
# Paid AI endpoint. Rendering /analyze/?s=… must not trigger a model call; a
# rendering crawler honours this for the fetches the page makes (2026-09-27:
# ~98% of anonymous X-Ray analyses were crawler renders of those links).
Disallow: /.netlify/functions/analyze-sentence
